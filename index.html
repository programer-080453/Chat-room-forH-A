<!-- JavaScript（ロジック・通信処理）-->
    <script type="module">
        //Firebase公式のライブラリ（SDK）読み込み
        import { initializeApp } from "https://gstatic.com";
        import { getFirestore, collection, addDoc, query, orderBy, onSnapshot, serverTimestamp } from "https://gstatic.com";

        // Firebaseプロジェクトの接続情報（※ご自身の情報に書き換えてください）
        const firebaseConfig = {
            apiKey: "YOUR_API_KEY",
            authDomain: "YOUR_AUTH_DOMAIN",
            projectId: "YOUR_PROJECT_ID",
            storageBucket: "YOUR_STORAGE_BUCKET",
            messagingSenderId: "YOUR_MESSAGING_SENDER_ID",
            appId: "YOUR_APP_ID"
        };

        // セキュリティ(合言葉)
        const SECRET_WORD = "CTS0718";
        // 入力ポップアップを表示して入力を促す
        const userInput = prompt("合言葉を入力してください：");

        // 入力された文字と、あらかじめ決めた合言葉が【一致しない】場合
        if (userInput !== SECRET_WORD) {
            alert("合言葉が違います。アクセスできません。");
            // 画面をエラー表示に強制書き換えして、以降のチャットプログラムを実行させない
           document.body.innerHTML = "<h2 style='text-align:center; margin-top:20vh; color:red;'>アクセス権限がありません</h2>";
        } else {
            // 合言葉完全一致チャットの処理を開始する
            startChat();
        }

        // チャットのメインプログラム（合言葉クリア後に実行される）
        function startChat() {
            // 1. 非表示にしていたチャット画面を表示（flex形式）にする
            const chatContainer = document.getElementById('chatContainer');
            chatContainer.style.display = 'flex';

            // 2. Firebaseとデータベース（Firestore）の接続・初期化
            const app = initializeApp(firebaseConfig);
            const db = getFirestore(app);

            // 3. ユーザーの識別（自分が送ったか、相手が送ったかを区別するため）
            if (!sessionStorage.getItem('userId')) {
                sessionStorage.setItem('userId', 'user_' + Math.random().toString(36).substring(2, 9));
            }
            const myUserId = sessionStorage.getItem('userId');

            // 4. HTMLの要素（入力欄やボタンなど）をJavaScriptで操作できるように取得
            const chatMessages = document.getElementById('chatMessages');
            const chatInput = document.getElementById('chatInput');
            const sendBtn = document.getElementById('sendBtn');

            // 5. 【送信処理】データベースにメッセージを保存する関数
            async function sendMessage() {
                const text = chatInput.value.trim(); // 前後の不要な空白を消す
                if (text === '') return;             // もし空っぽなら何もしない

                try {
                    // Firestoreの「messages」という名前のコレクション（データ置き場）にデータを追加
                    await addDoc(collection(db, "messages"), {
                        userId: myUserId,            // 送信者のID
                        text: text,                  // メッセージ本文
                        timestamp: serverTimestamp() // サーバー側で判定された正確な送信時刻
                    });
                    chatInput.value = ''; // 無事に送信できたら入力欄を空にする
                } catch (e) {
                    console.error("送信エラーが発生しました: ", e);
                }
            }

            // 6. 【受信処理】データベースの変更をリアルタイムで監視する処理
            const q = query(collection(db, "messages"), orderBy("timestamp", "asc"));

            onSnapshot(q, (querySnapshot) => {
                chatMessages.innerHTML = ''; // 過去の表示を一度全部クリア（二重表示を防ぐため）

                // データベースにあるメッセージを1件ずつ順番にループ処理
                querySnapshot.forEach((doc) => {
                    const data = doc.data();
                    // データベース側で時間の書き込みが完了（確定）するまでは処理をスキップする
                    if (!data.timestamp) return;

                    // --- 時間データの変換処理 ---
                    const date = data.timestamp.toDate();                  // JavaScriptのDate型に変換
                    const hours = String(date.getHours()).padStart(2, '0'); // 「時」を取得して常に2桁にする
                    const minutes = String(date.getMinutes()).padStart(2, '0'); // 「分」を取得して常に2桁にする
                    const timeString = `${hours}:${minutes}`;              // 「13:45」のような文字列を作る

                   
        // ① 外枠（ラッパー）を作る
                    const wrapperDiv = document.createElement('div');
                    wrapperDiv.classList.add('message-wrapper');

                    // ② 吹き出し（本文）を作る
                    const messageDiv = document.createElement('div');
                    messageDiv.classList.add('message');
                    messageDiv.textContent = data.text;

                    // ③ 時間表示部分を作る
                    const timeDiv = document.createElement('div');
                    timeDiv.classList.add('message-time');
                    timeDiv.textContent = timeString;

                    // 自分の発言か、相手の発言かで見た目（左右）を切り替える
                    if (data.userId === myUserId) {
                        wrapperDiv.classList.add('self');
                        messageDiv.classList.add('self');
                    } else {
                        wrapperDiv.classList.add('other');
                        messageDiv.classList.add('other');
                    }

                    // パーツを合体させて画面に追加
                    wrapperDiv.appendChild(messageDiv);
                    wrapperDiv.appendChild(timeDiv);
                    chatMessages.appendChild(wrapperDiv);
                });

                // 新しいメッセージが届いたら、常に一番下まで自動でスクロールさせる
                chatMessages.scrollTop = chatMessages.scrollHeight;
            });

            // 7. 【イベント登録】ボタンのクリックやEnterキーが押された時の設定
            sendBtn.addEventListener('click', sendMessage);
            chatInput.addEventListener('keypress', (e) => {
                if (e.key === 'Enter') sendMessage();
            });
        }
    </script>
